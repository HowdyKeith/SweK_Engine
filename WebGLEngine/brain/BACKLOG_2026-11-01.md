import * as TA from "../brain/subTactics.mjs";

// PROBE 1: 12 duels, the default tactician and the PD autopilot on both sides, seeds 90001..90012: time, endings, torpedoes, and the share of frames each boat has the other in its baffles.
{
    const seeds = Array.from({ length: 12 }, (_, i) => 90001 + i);
    let t0 = Date.now(), n = 0, baf = [0, 0], alive = [0, 0], ends = { sunk: 0, draw: 0 }, fired = 0, hits = 0, dur = [];
    for (const sd of seeds) {
        const sc = D.scenario(sd, "patrol"), trace = [];
        const r = D.duel(D.sideHand(TA.THETA_DEFAULT)(), sc, { opponent: D.sideHand(TA.THETA_DEFAULT)(), trace });
